/* ===================================================== CANLI DÖVİZ KURU ===================================================== */ async function loadCurrency(){ const usdEl = document.getElementById("usd"); const eurEl = document.getElementById("eur"); const gbpEl = document.getElementById("gbp"); try{ usdEl.textContent = "Güncelleniyor..."; eurEl.textContent = "Güncelleniyor..."; gbpEl.textContent = "Güncelleniyor..."; const response = await fetch( "https://open.er-api.com/v6/latest/TRY?nocache=" + Date.now(), { cache: "no-store" } ); if(!response.ok){ throw new Error("Kur verisi alınamadı"); } const data = await response.json(); if(!data.rates){ throw new Error("Kur verisi bulunamadı"); } /* TRY bazından USD/EUR/GBP değerini TL'ye çevir */ const usd = 1 / data.rates.USD; const eur = 1 / data.rates.EUR; const gbp = 1 / data.rates.GBP; usdEl.textContent = usd.toLocaleString("tr-TR", { minimumFractionDigits:2, maximumFractionDigits:2 }) + " TL"; eurEl.textContent = eur.toLocaleString("tr-TR", { minimumFractionDigits:2, maximumFractionDigits:2 }) + " TL"; gbpEl.textContent = gbp.toLocaleString("tr-TR", { minimumFractionDigits:2, maximumFractionDigits:2 }) + " TL"; console.log( "Döviz kurları güncellendi:", new Date().toLocaleTimeString("tr-TR") ); }catch(error){ console.error("Kur güncelleme hatası:", error); usdEl.textContent = "Alınamadı"; eurEl.textContent = "Alınamadı"; gbpEl.textContent = "Alınamadı"; } }
